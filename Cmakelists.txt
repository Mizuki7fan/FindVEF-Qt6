cmake_minimum_required(VERSION 3.7)
project(FindVEF-Qt6)

#设置不生成MinSizeRel;RelWithDebInfo
set(CMAKE_CONFIGURATION_TYPES "Debug;Release")

if(CMAKE_BUILD_TYPE STREQUAL "")
    message(FATAL_ERROR CMAKE_BUILD_TYPE的值为空)
endif()

include(${CMAKE_CURRENT_LIST_DIR}/script/cmake/vars.cmake)
include(${CMAKE_CURRENT_LIST_DIR}/script/cmake/paths.cmake)

# 设置不同配置的输出目录, 在使用msvc的时候应该这样写
# 设置生成exe的存放位置
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_DEBUG "${CMAKE_SOURCE_DIR}/run/bin_Debug")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_SOURCE_DIR}/run/bin_Release")
set(CMAKE_DEBUG_POSTFIX "d")#设置生成lib的Debug后缀
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY_DEBUG "${CMAKE_SOURCE_DIR}/run/bin_Debug")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY_RELEASE "${CMAKE_SOURCE_DIR}/run/bin_Release")

add_definitions(/D_USE_MATH_DEFINES)
SET(CMAKE_CXX_STANDARD 20)
SET(CMAKE_CXX_STANDARD_REQUIRED ON)

# 引入 Qt6 库配置文件
include(${CMAKE_CURRENT_LIST_DIR}/script/cmake/config_qt6.cmake)

# 引入 Boost 库配置文件

include(${CMAKE_CURRENT_LIST_DIR}/script/cmake/config_boost.cmake)

include_directories(SYSTEM ${CGAL_PATH}/include)
include_directories(SYSTEM ${EIGEN_PATH})# head-only库
include_directories(SYSTEM ${gltf_PATH})# head-only库

# 主文件
file (GLOB MAIN_FILE
    src/main.cpp
)

# 核心模块文件
file (GLOB CORE_FILES
    src/core/geometry/io/*
    src/core/geometry/*
)

# UI模块文件
file (GLOB UI_FILES
    src/ui/windows/*.cpp
    src/ui/windows/*.h
    src/ui/widgets/*.cpp
    src/ui/widgets/*.h
)


# 工具模块文件
file (GLOB UTILS_FILES
    src/utils/*.cpp
    src/utils/*.h
)

# 渲染通用文件
file (GLOB RENDER_COMMON_FILES
    src/render/common/*.cpp
    src/render/common/*.h
)

# 根据选项包含渲染文件
set(RENDER_FILES ${RENDER_COMMON_FILES})

# 初始化由后端导出的链接库聚合变量
set(RENDER_LINK_LIBRARIES "")

# 按开关条件包含三种后端
if(USE_OPENGL1_0)
    include(${CMAKE_CURRENT_LIST_DIR}/script/cmake/config_opengl1.cmake)
endif()

if(USE_OPENGL3_0)
    include(${CMAKE_CURRENT_LIST_DIR}/script/cmake/config_opengl3.cmake)
endif()

if(USE_VULKAN)
    include(${CMAKE_CURRENT_LIST_DIR}/script/cmake/config_vulkan.cmake)
endif()

# 引入 FBX SDK 配置文件（需要在文件列表定义之后，add_executable之前）
if(SUPPORT_FBX)
    include(${CMAKE_CURRENT_LIST_DIR}/script/cmake/config_fbx.cmake)
endif()

add_executable(${PROJECT_NAME} 
    ${MAIN_FILE}
    ${CORE_FILES}
    ${UI_FILES}
    ${UTILS_FILES}
    ${RENDER_FILES}
)

# 库链接配置
set(LINK_LIBRARIES
    Qt6::Widgets Qt6::Core Qt6::Gui
    ${Boost_LIBRARIES}
)

# 追加由渲染后端产生的链接库
list(APPEND LINK_LIBRARIES ${RENDER_LINK_LIBRARIES})

target_link_libraries(${PROJECT_NAME} PRIVATE ${LINK_LIBRARIES})

# 配置 FBX SDK（如果启用）
if(SUPPORT_FBX)
    configure_fbx_target(${PROJECT_NAME})
endif()

# 配置 Qt6 目标属性
configure_qt6_target(${PROJECT_NAME})

# 设置编译器标志, 这句话要在target_link_libraries后面
if (CMAKE_CXX_COMPILER_ID STREQUAL "MSVC")
    # 移除默认的警告级别，然后设置/W4，避免D9025警告
    string(REGEX REPLACE "/W[0-4]" "" CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS}")
    target_compile_options(${PROJECT_NAME} PRIVATE /W4)# /W4 较高的警告等级; 暂时禁用/WX以避免警告导致编译失败
    # 禁用 unreachable code 警告（C4702）
    target_compile_options(${PROJECT_NAME} PRIVATE /wd4702)
endif()

# 配置 Qt6 部署
configure_qt6_deployment(${PROJECT_NAME})

# 复制 FBX SDK 相关的 DLL 文件（如果启用）
if(SUPPORT_FBX)
    configure_fbx_dll_copy(${PROJECT_NAME})
endif()
